% Devanagari font handling for xelatex.
% TEMPLATE FILE — processed by build_book.py at build time.
% The token Adobe Devanagari is replaced with the value of `devanagarifont`
% from as_book.yaml; the rendered file is written to build/devanagari-preamble.tex
% and included via pandoc's --include-in-header flag.
%
% Strategy: explicit per-run font wrapping (no automatic char-class transitions).
%   - mainfont (set via as_book.yaml) handles Latin + IAST diacritics.
%   - build_book.py's assemble step wraps every Devanagari run in raw-LaTeX
%     `{\devanagarifont …}` so the font switch is unconditional inside the
%     wrap group and reverts cleanly at the closing brace.
%   - ucharclasses was tried earlier and removed: it routes the Devanagari
%     danda U+0964 (and other Devanagari punctuation codepoints) through a
%     separate character-class transition that overrides the wrap-group's
%     font selection, producing missing-glyph boxes for the danda even when
%     the run is wrapped. Without ucharclasses the wrap is sufficient.
%
% To change the Devanagari font, edit `devanagarifont` in as_book.yaml.
% Do NOT hardcode font names in this file.

\usepackage{fontspec}

\newfontfamily{\devanagarifont}[Script=Devanagari]{Adobe Devanagari}

% Pre-warm the Devanagari font at document start so the first wrap-group
% transition does not leave its first glyph in the outgoing font.
\AtBeginDocument{\setbox0=\hbox{\devanagarifont क}}
